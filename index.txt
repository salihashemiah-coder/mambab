  try{P=JSON.parse(LS.P||'[]')}catch(e){P=[]}
  if(!P.length)await refreshProducts();
  PM={};P.forEach(p=>PM[p[0]]=p);
  $('hUser').textContent='👤 '+user;show('vHome');paintHome();flush();

  stage=s;clearPick();$('cTitle').textContent='شمارش '+['','اول','دوم','سوم'][s]+(set.closed[s]?' 🔒':'');
  NEED=[];NM={};
  if(s>1){try{NEED=await run('getNeedList',s);NEED.forEach(n=>NM[n[0]]=n)}catch(e){return err(e)}}
  const cats={};
  if(s===1)P.forEach(p=>cats[p[2]]=1);else NEED.forEach(n=>{if(!n[2])cats[n[1]]=(cats[n[1]]||0)+1});
  $('cat').innerHTML=(s===1?'<option value="">همه گروه‌ها</option>':'<option value="">انتخاب گروه کالا</option>')+
    Object.keys(cats).sort().map(c=>`<option value="${c}">${c}${s>1?' ('+cats[c]+' باقی‌مانده)':''}</option>`).join('');

function renderSearch(){
  const q=norm($('q').value).trim(),cat=$('cat').value;if(!q&&!(stage>1&&cat)){$('res').innerHTML='';return}
  const toks=q.split(/\s+/).filter(Boolean);let out=[];
  for(const p of P){
    if(stage>1){const n=NM[p[0]];if(!n||n[2]||(cat&&p[2]!==cat))continue}else if(cat&&p[2]!==cat)continue;
    const h=norm(p[1]+' '+p[0]);if(toks.every(t=>h.includes(t))){out.push(p);if(out.length>=30)break}
  }
  $('res').innerHTML=out.length?out.map(p=>`<div class="li" onclick="pickP('${p[0]}')"><b>${p[1]}</b><div class="m">${p[0]}</div></div>`).join(''):'<div class="m">موردی پیدا نشد</div>';

  if(!sel)return;if(set.closed[stage])return toast('این مرحله بسته است');

  clearPick();$('q').value='';$('res').innerHTML='';renderMine();flush();
}

  if(flushing||!queue().length)return;flushing=true;
  try{for(const st of [1,2,3]){const items=queue().filter(e=>e.stage===st);if(!items.length)continue;
    const done=await run('addCounts',st,user,items.map(({id,barcode,expr})=>({id,barcode,expr})));
    setQ(queue().filter(e=>!(e.stage===st&&done.includes(e.id))))}}catch(e){if(e&&e.message&&/بسته/.test(e.message))err(e)}
  flushing=false;paintHome();if(stage)renderMine();
}

}

  scanner.start({facingMode:'environment'},{fps:10,qrbox:{width:300,height:150}},t=>{stopScan();gotCode(t)},()=>{}).catch(()=>{scanner=null;toast('دوربین زنده مجاز نیست؛ از «عکس بارکد» استفاده کنید')});

  if(!PM[t])toast('بارکد '+t+' در لیست نیست');else pickP(t)}
async function scanPhoto(inp){
  const f=inp.files[0];inp.value='';if(!f)return;let t=null;toast('در حال خواندن...');
  try{if('BarcodeDetector' in window){const r=await new BarcodeDetector().detect(await createImageBitmap(f));if(r.length)t=r[0].rawValue}}catch(e){}
  if(!t){try{const sc=new Html5Qrcode('reader');t=await sc.scanFile(f,false);sc.clear()}catch(e){}}
  if(t)gotCode(t);else toast('بارکد خوانده نشد؛ نزدیک‌تر و واضح‌تر عکس بگیرید')
